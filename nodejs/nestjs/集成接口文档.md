# 集成接口文档

## 配置 Swagger 文档
* 安装
  ```bash
    npm install @nestjs/swagger
    # 若使用 JWT 认证，还需安装相关包
    npm install @nestjs/passport passport-jwt @types/passport-jwt
  ```

* 配置 Swagger 文档
  ```ts
    // doc.ts
    import { SwaggerModule, DocumentBuilder } from '@nestjs/swagger';

    export const generateDocument = (app) => {
        const options = new DocumentBuilder()
            .setTitle('nestjs-api接口文档')
            .setDescription('接口文档')
            .setVersion('1.0')
            .setContact('xintairuan', 'http://www.example.cn', 'qinmao@example.com')
            // 1. 添加 Bearer 认证
            .addBearerAuth(
            {
                type: 'http',
                scheme: 'bearer',
                bearerFormat: 'JWT',
                name: 'Authorization',
                description: 'Enter JWT token',
                in: 'header',
            },
            'BearerAuth',
            )
            // 2. 全局启用 BearerAuth 授权，避免在每个控制器或方法上重复添加 @ApiBearerAuth('BearerAuth')
            .addSecurityRequirements('BearerAuth', []) // 所有接口默认需要 BearerAuth 授权
            // 3. 添加一个空安全方案
            // 4. 对于不需要授权的接口（如登录、注册），使用 @ApiSecurity('public') 标记它们
            .addSecurity('public', { type: 'apiKey', in: 'header', name: 'NoAuth' })
            .build();

        const documentFactory = SwaggerModule.createDocument(app, options, {
            // 忽略通过 setGlobalPrefix() 设置的路由的全局前缀
            // ignoreGlobalPrefix: true
            // include?: Function[];
        });

        SwaggerModule.setup('swagger', app, documentFactory, {
            // http://localhost:3000/swagger/json
            jsonDocumentUrl: 'swagger/json',
            yamlDocumentUrl: 'swagger/yaml',
            swaggerOptions: {
            persistAuthorization: true, // 使授权信息在页面刷新时保持
            defaultModelsExpandDepth: -1, // 隐藏 Schemas 部分
            docExpansion: 'none', // 默认折叠所有 API 路径
            deepLinking: true, // 启用深度链接
            displayOperationId: true, // 显示 operationId
            filter: true, // 启用过滤器功能
            maxDisplayedTags: 10, // 设置显示的最大标签数量
            showExtensions: true, // 显示扩展
            showCommonExtensions: true, // 显示常见扩展
            // 兼容 省略输入 Bearer + 空格
            requestInterceptor: (req) => {
                const token = req.headers.Authorization as string;
                if (token && !token.startsWith('Bearer ')) {
                req.headers.Authorization = `Bearer ${token}`;
                }
                return req;
            },
            },
            customCss: `
            .swagger-ui .topbar { background-color: #4CAF50; }
            .swagger-ui .topbar .topbar-wrapper .link span { color: white; }
            .swagger-ui .info { margin: 20px 0; }
            .swagger-ui .info .title { font-size: 24px; font-weight: bold; }
            .swagger-ui .info .description { font-size: 14px; }
            `,
            customSiteTitle: 'NestJS API 文档',
        });
    };

    // main.ts
    // 开发环境创建文档
    const env = getConfig('ENV') as string;
    if (env === 'dev') {
        generateDocument(app);
    }
  ```

* 标记 不需要 swagger 授权的接口
  ```ts
    import { ApiSecurity, ApiTags } from '@nestjs/swagger';

    @ApiTags('Auth')
    @Controller('auth')
    export class AuthController {
        @Post('login')
        @ApiSecurity('public') // 覆盖全局授权配置
        async login() {
            // 登录逻辑（无需授权）
        }
    }
  ```

* 配置响应参数
  ```ts
    import { applyDecorators, Type } from '@nestjs/common';
    import {
    ApiExtraModels,
    ApiResponse,
    getSchemaPath,
    ApiProperty,
    } from '@nestjs/swagger';

    /** 通用响应结构 */
    export class ApiRes<T = any> {
    @ApiProperty({
        example: '200',
        description: '接口响应编码:200 成功、500 服务类错误',
    })
    code: number;

    @ApiProperty({ description: '接口响应数据' })
    result: T;

    @ApiProperty({ description: '接口响应消息' })
    message: string;
    }

    /** inline schema 的选项 */
    export type ApiResInlineOptions = {
        schema: any; // 必须提供的 inline swagger schema
        dataProp?: string; // result 属性名，默认 'result'
    };

    /**
    * 为接口添加通用响应结构的 Swagger 文档
    * @param arg DTO class 或 inline schema 选项
    * @Examples
    * ```typescript
    * // 1. 使用 DTO class
    * @ApiResWith(UserDto)
    * @Post('createUser')
    * async createUser(...) { ... }
    *
    * // 2. 使用 inline schema
    * @ApiResWith({
    *   schema: {
    *     type: 'object',
    *     properties: {
    *       accessToken: { type: 'string' },
    *     },
    *   },
    *   dataProp: 'data', // 可选，默认为 'result'
    * })
    * @Post('login')
    * async login(...) { ... }
    * ```
    * @returns
    */
    export function ApiResWith(arg?: Type<unknown> | ApiResInlineOptions) {
    const dataPropDefault = 'result';

    // 如果传 DTO class
    if (typeof arg === 'function') {
        const model = arg;
        const schema = {
        allOf: [
            { $ref: getSchemaPath(ApiRes) },
            { properties: { [dataPropDefault]: { $ref: getSchemaPath(model) } } },
        ],
        };
        return applyDecorators(
        ApiExtraModels(ApiRes, model),
        ApiResponse({ schema }),
        );
    }

    // inline schema 路径
    const opts = (arg as ApiResInlineOptions) || {
        schema: { type: 'object' },
    };
    const dataProp = opts.dataProp ?? dataPropDefault;
    const inlineSchema = opts.schema ?? { type: 'object' };

    const schema = {
        allOf: [
        { $ref: getSchemaPath(ApiRes) },
        { properties: { [dataProp]: inlineSchema } },
        ],
    };

    return applyDecorators(ApiExtraModels(ApiRes), ApiResponse({ schema }));
    }

  ```
